The Custom Way

Without a ModelForm, an edit page copies values by hand, in both directions. This illustration shows the two ways that goes: copying everything that was sent, and copying carefully. Run it with any Python, and do not add it to the project:

illustration_only/by_hand.py (not part of the project)

STORED = {'title': 'Online Django clinic',
          'status': 'draft', 'capacity': 50}
SENT = {'title': 'Clinic', 'capacity': '1',
        'status': 'published'}


def update_everything(row, sent):
    for name, value in sent.items():
        row[name] = value
    return row


def update_carefully(row, sent, places_taken):
    errors = {}
    row = dict(row, title=sent.get('title', '').strip())
    if not row['title']:
        errors['title'] = 'This field is required.'
    try:
        row['capacity'] = int(sent.get('capacity', ''))
    except ValueError:
        errors['capacity'] = 'Enter a whole number.'
    else:
        if row['capacity'] < places_taken:
            errors['capacity'] = 'Below the places taken.'
    return errors or row


print(update_everything(dict(STORED), SENT))
print(update_carefully(STORED, SENT, places_taken=2))
print(update_carefully(STORED, SENT | {'capacity': '40'},
                       places_taken=2))

Output

{'title': 'Clinic', 'status': 'published', 'capacity': '1'}
{'capacity': 'Below the places taken.'}
{'title': 'Clinic', 'status': 'draft', 'capacity': 40}

The first version does what the request asks: the title changes, the capacity becomes the text '1' instead of a number, and status becomes published, because the request said so. The careful version names the fields it allows, converts the capacity, and refuses a value below the places taken. It is already twenty lines for two fields, and it still has no labels, no redisplay, no times, no zones, and no check of who is asking. Every rule is a copy of one the model already holds. The Django Way below keeps the careful version’s ideas and lets the form and the model do the copying.

../practice/lifecycle.py

import setup_django
from events.forms import EventForm
from events.models import Event

calls = []


def traced(name, method):
    def wrapper(self, *args, **kwargs):
        calls.append(f'model {name}()')
        return method(self, *args, **kwargs)
    return wrapper


for name in ('clean_fields', 'clean', 'validate_unique',
             'validate_constraints'):
    setattr(Event, name, traced(name, getattr(Event, name)))


class TracedForm(EventForm):
    def clean_title(self):
        calls.append('form clean_title()')
        return self.cleaned_data['title']

    def clean(self):
        calls.append('form clean()')
        return super().clean()


clinic = Event.objects.get(title='Online Django clinic')
sent = {'title': 'Django clinic',
        'timezone': 'America/New_York',
        'starts_at': '2026-12-03T19:00',
        'ends_at': '2026-12-03T20:30',

print(*calls, sep='\n')

Every test stops at reverse('events:edit', ...), because the address does not exist yet. Start in events/models.py. Replace the method is_visible_to() of Event with these three methods:

events/models.py (is_visible_to, replaced)

    def has_organizer(self, user):
        """Return True if user is a current organizer."""
        if not user.is_authenticated:
            return False
        return self.organizer_assignments.filter(
            member=user, revoked_at__isnull=True
        ).exists()

    def is_visible_to(self, user):
        """Whether user may see this event.

        Published: everyone. Otherwise: organizers and
        holders of the view_event permission.
        """
        if self.status == self.Status.PUBLISHED:
            return True
        if user.has_perm('events.view_event'):
            return True
        return self.has_organizer(user)

    def can_be_edited_by(self, user):
        """Holders of change_event who organize it."""
        return (user.has_perm('events.change_event')
                and self.has_organizer(user))

has_organizer() is the question Chapter 24 asked inside is_visible_to(), moved into its own method so that both rules share it, as Chapter 22’s “fat models, thin views” asks. can_be_edited_by() answers P1 of this chapter’s plan: the permission and a current assignment, both. A visitor holds no permissions, so has_perm() returns false for them without asking the database, and and then skips the second question.

events/views.py (add at the end)

@login_required
def event_edit(request, event_id):
    """Edit the event, for those who may change it."""
    event = get_object_or_404(Event, pk=event_id)
    if not event.is_visible_to(request.user):
        raise Http404('No event matches the given query.')
    if not event.can_be_edited_by(request.user):
        raise PermissionDenied
    form = EventForm(instance=event)
    return render(request, 'events/edit.html', {
        'form': form,
        'event': event,
    })

login_required sends a visitor to sign in. The view then asks the two questions in order: may this person see the event at all, and if so, may they edit it. class EditCsrfTests(TestCase):
    """An edit without the CSRF token is refused."""
